    private static void initializeboard(int[][] board) {
        playerBoard = new int[ROWS][COLS];
        for (int r = 0; r < ROWS; r++) {
            for (int c = 0; c < COLS; c++) {
                playerBoard[r][c] = board[r][c];
            }
        }
    }

    private static void initializeLegalMoves() {
        legalMoves = new ArrayList[ROWS][COLS];
        for (int r = 0; r < ROWS; r++) {
            for (int c = 0; c < COLS; c++) {
                legalMoves[r][c] = new ArrayList(COLS);
                for (int value = 1; value <= COLS; value++) {
                    if (playerBoard[r][c] == 0) {
                        legalMoves[r][c].add(value);
                    }
                }

            }
        }
    }

    public static boolean isValid(int num, int row, int column) {
//        System.out.println("box" + checkBox(num,row,column));
//        System.out.println("row" + checkRow(num,row));
//        System.out.println("column" + checkColumn(num,column));
        return (!checkBox(num, row, column) && !checkRow(num, row) && !checkColumn(num, column) && (playerBoard[row][column] == 0));
    }

    public static boolean checkBox(int num, int row, int column) {
        boolean isInBox = false;
        int rowMod = row - row % 3;
        int columnMod = column - column % 3;
        for (int r = rowMod; r < (rowMod + 3); r++) {
            for (int c = columnMod; c < (columnMod + 3); c++) {
                if (num == playerBoard[r][c]) {
                    isInBox = true;
                }
            }
        }
        //System.out.println("rowMod" + rowMod);
        //System.out.println("columnMod" + columnMod);
        return isInBox;

    }

    public static boolean checkRow(int num, int row) {
        boolean isInRow = false;
        for (int c = 0; c < COLS; c++) {
            if (num == playerBoard[row][c]) {
                isInRow = true;
            }
        }
        return isInRow;
    }

    public static boolean checkColumn(int num, int column) {
        boolean isInColumn = false;
        for (int r = 0; r < ROWS; r++) {
            if (num == playerBoard[r][column]) {
                isInColumn = true;
            }
        }
        return isInColumn;
    }

    public static ArrayList<Integer> removeDup(ArrayList<Integer> orginal, ArrayList<Integer> remove) {
        for (int r = 0; r < remove.size(); r++) {
            for (int o = 0; o < orginal.size(); o++) {
                if (remove.get(r) == orginal.get(o)) {
                    orginal.remove(o);
                    o--;
                }
            }
        }
        return orginal;
    }

    public static void oneInBox() {
        for (int r1 = 0; r1 < ROWS; r1 += 3) {

                    int times = 0;
                    int row = 0;
                    int col = 0;
                    for (int r2 = r1; r2 < r1 + 3; r2++) {
                        for (int c2 = c1; c2 < c1 + 3; c2++) {
                            for (int i = 0; i < legalMoves[r2][c2].size(); i++) {
                                if (legalMoves[r2][c2].get(i) == n) {
                                    times++;
                                    row = r2;
                                    col = c2;
                                }
                            }

                        }
                    }
                    if (times == 1 && isValid(n, row, col)) {
                        playerBoard[row][col] = n;
                        CHANGED++;
                    }
                }
            }
        }
    }

            int col = input.nextInt() - 1;

            //System.out.println("row " + row + "col "+ col + "num " + num);

            remove(board, row, col);
        } else if (playerInput.equals("clear")) {
            clear(board);
            printCardBlanks();
        }
        if (!complete()) {
            if (!playerInput.equals("end")) {
                playerInput(input, board);
            }
        }
    }
